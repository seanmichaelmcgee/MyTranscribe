# Optional recognisers for Options -> Recogniser. The standard install (requirements-1060.txt)
# needs none of this: Whisper works without it.
#
# Install into the SAME environment as the app, after requirements-1060.txt:
#
#   1. PyTorch for MedASR (CUDA 12.6 build, about 2.5 GB):
#        pip install --only-binary=:all: torch==2.13.0 --index-url https://download.pytorch.org/whl/cu126
#   2. pip install -r requirements-engines.txt
#   3. faster-whisper asks for the CPU "onnxruntime"; onnxruntime-gpu below replaces it (same module).
#      If pip put both in, remove the CPU one:  pip uninstall onnxruntime  then reinstall onnxruntime-gpu.
#
# Everything here is the CUDA 12 / cuDNN 9 family, the same as the app's own GPU libraries. Do not
# mix in a CUDA 13 build of onnxruntime-gpu (1.24 and later): two cuDNN versions cannot share a process.
#
# Models are separate from packages:
#   Parakeet / Canary: fetched from Hugging Face on first use, after the app asks (CC BY 4.0, no account).
#   MedASR: gated by Google. Accept the terms at https://huggingface.co/google/medasr with your own
#           account, run "hf auth login", then  python scripts\download_medasr.py  (docs/MEDASR_TRIAL.md).

# Parakeet and Canary (ONNX exports run by onnx-asr)
onnx-asr[hub]==0.12.0
onnxruntime-gpu==1.23.2

# MedASR (with torch from step 1)
transformers==5.18.0
safetensors==0.8.0
sentencepiece==0.2.1

truststore>=0.10
